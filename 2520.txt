/* 2520 Count the Digits That Divide a Number */



class Solution {
public:
    int countDigits(int num) {
        int cnt=0;
        int num_old = num;
        while(num !=0){
            if(num_old % (num%10) == 0)
                cnt++;

            num = num/10;
        }
        return cnt;
    }
};
