/* Power of 4 */

/* Whenever you see:
	​
log(BASE) / log(NUMBER)
	​
*/


class Solution {
public:
    bool isPowerOfFour(int n) {
        long pow=1;
        while(pow<n){
            pow = pow*4;
        }
        return (pow == n);
    }
};


/* Solution 2 */

class Solution {
public:
    bool isPowerOfFour(int n) {
        while(n >0){
            if(n%4 == 0)
                n = n/4;
            else  
                break;
        }
        return (n == 1);
    }
};


/* Solution 3 */

class Solution {
public:
    bool isPowerOfFour(int n) {
        int cnt=0;

        while(n>0){
           
            cnt+= n%4;
            n = n/4;
        }
        return (1 == cnt);
    }
};


/* Solution 4 --- Recursion */

class Solution {
public:
    bool isPowerOfFour(int n) {
        if(n==1)
            return true;
        
        if(n <=0 || n%4 !=0)
            return false;
        
        return isPowerOfFour(n/4);
    }
};
