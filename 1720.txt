/* Decide XORed Array */


class Solution {
public:
    vector<int> decode(vector<int>& encoded, int first) {
        vector<int> encode;
        encode.push_back(first);
        for(int i=0;i<encoded.size();i++){
            first = first ^ encoded[i];
            encode.push_back(first);
        }

        return encode;
    }
};
