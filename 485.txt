/* 485 Max Consecutive Ones */

class Solution {
public:
    int findMaxConsecutiveOnes(vector<int>& nums) {
        int prev_cnt=0, count=0;
        for(int i=0;i<nums.size();i++){
            if(nums[i] == 1)
                count++;
            else{
                if(prev_cnt < count)
                    prev_cnt = count;
                count=0;
            }
                
        }
        return (prev_cnt > count? prev_cnt:count);
    }
};
