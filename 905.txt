/* Sort Array by Parity */

class Solution {
public:
    vector<int> sortArrayByParity(vector<int>& nums) {
        int no_even =-1;
        for(int i=0;i<nums.size();i++){
            if(nums[i] %2 == 0)
            {
                swap(nums[++no_even],nums[i]);
            } 
        }
        return nums;
    }
};
