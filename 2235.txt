#include <iostream>
using namespace std;

int add(int a, int b){
    //return a+b;
    
    if(b>=0){
        while(b--){
            a++;
        }
    }
    else{
        while(b++){
            a--;
        }
    }
    return a;
}
int main() {
     
     int result = add(5,6);
     cout<<result;

    return 0;
}

class Solution {
public:
    int sum(int num1, int num2) {
        int carry=-1,sum=0;
        while(carry!=0){
            sum = num1 ^ num2;
            carry = (num1 & num2)<<1;

            num1 = sum;
            num2 = carry;
        }
        return sum;
    }
};
