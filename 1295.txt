/* Find Numbers with Even Number of Digits */

class Solution {
public:
    bool isDigitEven(int no){
        int cnt=0;
        string ss = to_string(no);

        for(char c : ss){
            cnt++;
        }

        if(cnt%2==0)
            return true;
        else
            return false;
    }
    int findNumbers(vector<int>& nums) {
        int count=0;

        for(int i=0;i<nums.size();i++){
            if(isDigitEven(nums[i]))
                count++;
        }

        return count;
    }
};

class Solution {
public:
    int findNumbers(vector<int>& nums) {
        int count = 0;

        for (int i = 0; i < nums.size(); i++) {
            string ss = to_string(nums[i]);

            if (ss.size() % 2 == 0)
                count++;
        }

        return count;
    }
};

        int count = 0;

        for(int i=0;i<nums.size();i++){
            if((nums[i]>=10 && nums[i]<= 99) || (nums[i] == 100000) || (nums[i]>= 1000 && nums[i]<= 9999))
