/* 1365. How Many Numbers Are Smaller Than the Current Number */


/* Brute Force */

class Solution {
public:
 vector<int> smallerNumbersThanCurrent(vector<int>& nums) {
        
        int n = nums.size();
        vector<int> ans(n,0);
        
        for(int i=0;i<n;i++){
            for(int j=0;j<n;j++){
                if(nums[j]<nums[i])
                    ans[i]++;
            }
        }

        return ans;
    }  
};


class Solution {
public:
    vector<int> smallerNumbersThanCurrent(vector<int>& nums) {
        
        vector<int> ans;
        
        for(int i=0;i<nums.size();i++){
            int cnt=0;
            for(int j=0;j<nums.size();j++){
                if(i==j)
                    continue;
                
                if(nums[i]>nums[j])
                     cnt++;
            }
            ans.push_back(cnt);
        }

        return ans;
    }
};



/* Recursive Approach */

class Solution {
public:
    int countNo(vector<int>& nums,int index,int no){
        if(index == nums.size())
            return 0;

        int val = 0;

        if(nums[index] < no)
            val =1;
        
        return val + countNo(nums, ++index, no);
    }
 vector<int> smallerNumbersThanCurrent(vector<int>& nums) {
        
        vector<int> ans;

        for(int i=0;i<nums.size();i++){
           ans.push_back(countNo(nums,0,nums[i]));
        }
        return ans;
    }
};


/* Sort and hashmap */

class Solution {
public:
 vector<int> smallerNumbersThanCurrent(vector<int>& nums) {
        
        unordered_map<int,int> sortedNums;
        vector<int> sorted = nums;
        sort(sorted.begin(),sorted.end());
        vector<int> ans;
        int cnt=0;
        for(int i=0;i<sorted.size();i++){
            
            if(i>0 && (sorted[i-1] == sorted[i])){
                cnt++;
                sortedNums.insert({sorted[i],(i-cnt)});
            }    
            else
            {
                sortedNums.insert({sorted[i],i});
                cnt=0;
            }
        }

        //insert into ans vector
        for(auto n: nums)
            ans.push_back(sortedNums[n]);



        return ans;
    }  
};
