/* Add Binary */

Solution : 1

class Solution {
public:
    string addBinary(string a, string b) {
        int i = a.size();
        int j = b.size();
        int carry=0;
        int sum = 0;string ans;
        while(i>0 || j>0 || carry){
             sum = carry;

             if(i>0){
                sum=sum+a[--i] - '0';
             }

             if(j>0){
                sum =sum+b[--j]-'0';
             }

             ans.push_back(sum%2 + '0');
             carry = sum/2;
        }

        reverse(ans.begin(),ans.end());
        return ans;
        
    }
};


Solution 2: Recursive

class Solution {
public:
    string add(string &a, string &b, int i, int j, int carry){

        int x = (i>=0 ? a[i]-'0':0);
        int y = (j>=0 ? b[j]-'0':0);

        if(i<0 && j<0 && carry ==0)
            return "";
        
        int bit = x ^ y ^ carry;
        carry = (x & y) | (x & carry) | (y & carry);
        
        char ch = bit + '0';

        return  add(a,b,--i,--j,carry) + ch;
    }

    string addBinary(string a, string b) {
       return add(a,b,a.size()-1,b.size()-1,0);
    }
};
