class Solution {
  public:
    bool isPalindrome(string& s) {
        // code here
        string old_s = s;
        int first=0,last=s.length()-1;
        
        while(first<last){
            swap(s[first],s[last]);
            first++; last--;
        }
        return (old_s == s);
    }
};
