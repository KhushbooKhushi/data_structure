/* https://www.geeksforgeeks.org/problems/largest-element-in-array4009/1 */


class Solution {
  public:
    int largest(vector<int> &arr) {
        // code here
        int largest=arr[0];
        for(int i=1;i<arr.size();i++){
            if(largest < arr[i])
                largest = arr[i];
        }
        
        return largest;
    }
};

class Solution {
  public:
    int largest(vector<int> &arr) {
        // code here
        sort(arr.begin(),arr.end());
        
        return arr[arr.size()-1];
    }
};
