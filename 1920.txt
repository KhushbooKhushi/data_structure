/* 1920. Build Array from Permutation */


class Solution {
public:
    vector<int> buildArray(vector<int>& nums) {
        vector<int> ans;

        for(int i=0;i<nums.size();i++)
            ans.push_back(nums[nums[i]]);

        return ans;
    }
};



/* Optimal Solution 


the key pattern to remember
	
	value = new * n + odd

is basically the same idea as
	
	number = quotient * divisor + Remainder

In code why not simply use
	
	nums[i] = nums[nums[i]] * n + nums[i];

Because nums[nums[i]] might already have been encoded
	
	 (nums[nums[i]] % n) * n + nums[i];

*/


class Solution {
public:
    vector<int> buildArray(vector<int>& nums) {

        int n = nums.size();
        for(int i=0;i<n;i++)
            nums[i]= (nums[nums[i]] % n) * n + nums[i];

        for(int i=0;i<n;i++)
           nums[i] = nums[i]/n;

        return nums;
    }
};
